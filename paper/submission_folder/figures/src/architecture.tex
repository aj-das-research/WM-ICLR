% transport window from head_data: #1 origin, #2 weights (49, max-normalised), #3 cell, #4 identity mark
\newcommand{\transportwindow}[4]{%
  \begin{scope}[shift={#1}]
    \foreach \v [count=\n from 0] in #2 {
      \pgfmathtruncatemacro{\col}{mod(\n,7)}\pgfmathtruncatemacro{\row}{6-int(\n/7)}
      \pgfmathsetmacro{\pct}{100*sqrt(\v)}
      \fill[oursgreen!\pct!white] ({\col*#3},{\row*#3}) rectangle ++(#3,#3);}
    \draw[white, line width=0.2pt, step=#3] (0,0) grid (7*#3,7*#3);
    \draw[oursgreen!70!black, line width=0.6pt] (0,0) rectangle (7*#3,7*#3);
    \ifnum#4=1 \draw[gateamber, line width=1.0pt] (3*#3,3*#3) rectangle ++(#3,#3); \fi
  \end{scope}}

\newcommand{\tensor}[5]{%
  \begin{scope}[shift={#1}]
    \fill[#5!55] (0,#2) -- ++(#3,0.75*#3) -- ++(#2,0) -- ++(0,-#2) -- ++(-#3,-0.75*#3) -- cycle;
    \fill[#5!80] (#2,0) -- ++(#3,0.75*#3) -- ++(0,#2) -- ++(-#3,-0.75*#3) -- cycle;
    \draw[white, line width=0.3pt] (0,#2) -- ++(#3,0.75*#3) -- ++(#2,0) -- ++(-#3,-0.75*#3);
    \draw[white, line width=0.3pt] (#2,0) -- ++(#3,0.75*#3) -- ++(0,#2);
    \node[inner sep=0pt, anchor=south west] at (0,0) {\includegraphics[width=#2cm,height=#2cm]{arch_assets/#4}};

  \end{scope}}
% transformer layer slab: #1 lower-left, #2 width, #3 height, #4 colour
\newcommand{\slab}[4]{%
  \begin{scope}[shift={#1}]
    \fill[#4!18, draw=#4!70, line width=0.5pt, rounded corners=1.5pt] (0,0) -- (#2,0) -- ++(0.28,0.22) -- ++(0,#3) -- ++(-#2,0) -- ++(-0.28,-0.22) -- cycle;
    \draw[#4!70, line width=0.5pt] (0,0) -- (0,#3) -- ++(0.28,0.22); \draw[#4!70, line width=0.5pt] (0,#3) -- ++(#2,0) -- ++(0.28,0.22);
    \draw[#4!70, line width=0.5pt] (#2,0) -- (#2,#3);
  \end{scope}}
% token lattice: #1 lower-left, #2 n, #3 spacing, #4 draw colour, #5 fill colour
\newcommand{\lattice}[5]{%
  \begin{scope}[shift={#1}]
    \fill[white, draw=#4!55, line width=0.4pt, rounded corners=1pt] (-0.5*#3,-0.5*#3) rectangle ({(#2-0.5)*#3},{(#2-0.5)*#3});
    \foreach \x in {1,...,#2}{\foreach \y in {1,...,#2}{\fill[#5] ({(\x-1)*#3},{(\y-1)*#3}) circle (0.028);}}
  \end{scope}}
% memory token strip: #1 lower-left, #2 comma list of 8 hex colours (real feature colours)

  \end{scope}}
