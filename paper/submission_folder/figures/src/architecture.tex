% feature-vector strip: #1 origin, #2 comma list of 12 hex colours, #3 cell width
\newcommand{\vecstrip}[3]{%
  \begin{scope}[shift={#1}]
    \foreach \c [count=\n from 0] in #2 {\definecolor{vc}{HTML}{\c}\fill[vc] (\n*#3,0) rectangle ++(#3,0.3);}
    \draw[ink!60, line width=0.5pt, rounded corners=1pt] (0,0) rectangle (12*#3,0.3);
  \end{scope}}
% transport window: #1 origin, #2 comma list of 49 weights in [0,1], #3 cell size, #4 identity mark (1/0)
\newcommand{\transportwindow}[4]{%
  \begin{scope}[shift={#1}]
    \fill[obsblue!10] (0,0) rectangle (9*#3,9*#3);
    \foreach \v [count=\n from 0] in #2 {
      \pgfmathtruncatemacro{\col}{mod(\n,7)}\pgfmathtruncatemacro{\row}{6-int(\n/7)}
      \pgfmathsetmacro{\pct}{100*sqrt(\v)}
      \fill[oursgreen!\pct!white] ({(\col+1)*#3},{(\row+1)*#3}) rectangle ++(#3,#3);}
    \draw[white, line width=0.25pt, step=#3] (0,0) grid (9*#3,9*#3);
    \draw[oursgreen!80!black, line width=0.8pt] (#3,#3) rectangle (8*#3,8*#3);
    \draw[ink!50, line width=0.5pt] (0,0) rectangle (9*#3,9*#3);
    \ifnum#4=1 \draw[dynorange, line width=1.1pt] (4*#3,4*#3) rectangle ++(#3,#3); \fi
  \end{scope}}
\newcommand{\tokentile}[3]{%
  \begin{scope}[shift={#1}]
    \fill[#3, rounded corners=1pt] (0,0) rectangle (#2,#2);
    \draw[white, line width=0.3pt, step=#2/6] (0,0) grid (#2,#2);
    \draw[ink!45, line width=0.5pt, rounded corners=1pt] (0,0) rectangle (#2,#2);
  \end{scope}}
