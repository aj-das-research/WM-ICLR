\node[font=\scriptsize\itshape, text=obsblue!80!black, anchor=west, align=left] at (5.75,3.55) {features of\\real patches};

% observed window features z_{0,j} along the band; size ~ transport weight
\coordinate (zi)  at (1.75,2.35);   % current patch (stay)
\coordinate (p1)  at (1.1,1.8);
\coordinate (p2)  at (2.55,2.98);

\coordinate (p6)  at (6.05,4.25);   % weight 0.18
\coordinate (p7)  at (6.9,4.95);
\coordinate (T)   at (5.19,3.93);   % transport = weighted mean (convex hull)
\coordinate (zg)  at (4.85,3.8);    % (1-g) z0 + g T, g = 0.9
\coordinate (zt)  at (5.35,4.6);    % true future: near pi(i), plus new content
\coordinate (zh)  at (5.3,4.45);    % ShiftWM forecast

% transport lines (width ~ weight)
\foreach \p/\w in {p1/0.25, p2/0.3, p3/0.35, p4/0.8, pi/2.2, p6/1.2, p7/0.3}{
  \draw[oursgreen, line width=\w pt, opacity=0.75] (\p) -- (T);}
% additive head: must synthesise the full difference vector (Prop. 2)
\draw[->, addred, line width=1.0pt] (zi) to[out=78, in=170] node[pos=0.42, above, sloped, font=\scriptsize, text=addred, fill=panelgrey, inner sep=1pt] {synthesise $\z_{0,\pi(i)}{-}\z_{0,i}$} (zt);
% recursive rollout: drifts step by step (Prop. 3)
\draw[arorange, line width=0.8pt, densely dashed, ->] (zi) -- (2.35,1.95) -- (2.95,1.95) -- (3.55,1.72) -- (4.15,1.72) -- (4.75,1.45) -- (5.35,1.4) -- (5.95,1.1);
\foreach \q in {(2.35,1.95),(2.95,1.95),(3.55,1.72),(4.15,1.72),(4.75,1.45),(5.35,1.4)}{\fill[arorange] \q circle (0.045);}
\node[font=\scriptsize, text=arorange, anchor=north west, align=left] at (3.0,1.3) {recursive: each step feeds\\its own error forward};

% points
\foreach \p/\r in {p1/0.06, p2/0.065, p3/0.07, p4/0.09, p6/0.105, p7/0.065}{\fill[obsblue, draw=white, line width=0.4pt] (\p) circle (\r);}
\fill[obsblue, draw=white, line width=0.6pt] (pi) circle (0.15);

\node[font=\scriptsize, text=subink, anchor=north] at ($(zi)+(0.08,-0.14)$) {$\z_{0,i}$ (stay)};
\node[font=\scriptsize, text=obsblue!85!black, anchor=north] at ($(pi)+(0.25,-0.17)$) {$\z_{0,\pi(i)}$};

% gate: from stay toward transport
\draw[gateamber, line width=0.9pt, densely dotted] (zi) -- (T);
\fill[gateamber] (zg) circle (0.055);
% correction: small residual for new content
\draw[->, black!70, line width=0.8pt] (zg) -- (zh);
\node[font=\scriptsize, text=black!75, anchor=east] at ($(zg)!0.5!(zh)+(-0.04,0.02)$) {$\mathbf{r}$};
% transport point, forecast, truth
\node[diamond, fill=oursgreen, draw=white, line width=0.5pt, inner sep=1.5pt] at (T) {};
\node[circle, fill=oursgreen!90!black, draw=white, line width=0.6pt, inner sep=2.1pt] at (zh) {};
\node[star, star points=5, star point ratio=2.3, fill=ink, inner sep=1.3pt] at (zt) {};
\node[font=\scriptsize, text=ink, anchor=south] at ($(zt)+(0.1,0.12)$) {true $\z_{k,i}$};
\node[font=\scriptsize\bfseries, text=oursgreen!75!black, anchor=west] at ($(zh)+(0.14,-0.02)$) {$\hat{\z}_{k,i}$};
\node[font=\scriptsize, text=oursgreen!80!black, anchor=north west] at ($(T)+(0.08,-0.08)$) {$\mathbf{T}_{k,i}$};

% inset: the image grid (window around patch i)

  \draw[subink, line width=1.1pt] (0.1+0.3*3,0.1+0.3*3) rectangle ++(0.3,0.3); % patch i (centre)
  \draw[->, oursgreen, line width=0.9pt] (0.1+0.3*5.5,0.1+0.3*4.5) to[out=200, in=60] (0.1+0.3*3.55,0.1+0.3*3.6);
  \node[font=\scriptsize, text=ink, anchor=north] at (1.15,0.03) {window $\mathcal{W}(i)$};
  \node[font=\tiny, text=subink] at (1.15,-0.3) {part at $\pi(i)$ moves into $i$};
